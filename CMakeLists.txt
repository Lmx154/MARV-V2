cmake_minimum_required(VERSION 3.28)

project(marv VERSION 0.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

set(MARV_TARGET "host" CACHE STRING "Build target class: host or m33")
set_property(CACHE MARV_TARGET PROPERTY STRINGS host m33)
if(NOT MARV_TARGET STREQUAL "host" AND NOT MARV_TARGET STREQUAL "m33")
  message(FATAL_ERROR "MARV_TARGET must be host or m33, got '${MARV_TARGET}'")
endif()

include(cmake/flags.cmake)
include(cmake/flight_targets.cmake)
include(cmake/reference_data.cmake)

add_subdirectory(fw/prim)
add_subdirectory(fw/types)
add_subdirectory(fw/hal)
add_subdirectory(fw/sched)
add_subdirectory(fw/params)
add_subdirectory(fw/mixer)
add_subdirectory(fw/rate)
add_subdirectory(fw/attitude)
marv_add_card_report(marv_params CARD "${MARV_VEHICLE_CARD}" BUDGET "${MARV_DESIGN_BUDGET}")
add_subdirectory(fw/compositions/l0)
add_subdirectory(fw/compositions/l2_open_loop)
add_subdirectory(fw/compositions/l4_rate_scripted)
add_subdirectory(fw/compositions/l5_attitude_scripted)
add_subdirectory(tests/regression/quad/L00/controls)

if(MARV_TARGET STREQUAL "host")
  include(cmake/googletest.cmake)
  add_subdirectory(fw/hal/sim)
  add_subdirectory(fw/sil)
  add_subdirectory(sim/null_plant)
  add_subdirectory(sim/l2_open_loop)
  add_subdirectory(sim/l4_rate_scripted)
  add_subdirectory(sim/l5_attitude_scripted)
  add_subdirectory(sim/plant)
  add_subdirectory(sim/gz)
  add_subdirectory(tests/unit)
  add_subdirectory(tests/regression)
  add_subdirectory(tests/regression/quad/L01/controls)
  add_subdirectory(tests/regression/quad/L05/controls)
  add_subdirectory(tools/gen)
  add_subdirectory(tools/ci)
endif()

marv_generate_g3_manifest()
